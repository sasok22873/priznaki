name: CI v2

on:
  push:
    branches: [main]
    paths:
      - "v2/**"
      - ".github/workflows/ci-v2.yml"
  pull_request:
    paths:
      - "v2/**"
  workflow_dispatch:

concurrency:
  group: ci-v2-${{ github.ref }}
  cancel-in-progress: true

defaults:
  run:
    working-directory: v2

env:
  DATABASE_URL: postgresql://pb:pb@localhost:5432/pb
  SUPABASE_URL: https://mbbetmtxawduvcsttiis.supabase.co
  VITE_SUPABASE_URL: https://mbbetmtxawduvcsttiis.supabase.co
  VITE_SUPABASE_ANON_KEY: ci-placeholder-anon-key

jobs:
  verify:
    name: Typecheck · tests · build
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4
        with:
          version: 9.12.0
          package_json_file: v2/package.json

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: pnpm
          cache-dependency-path: v2/pnpm-lock.yaml

      - name: Install
        run: pnpm install --frozen-lockfile

      - name: Prisma validate
        run: pnpm --filter @pb/api db:validate

      - name: Prisma generate
        run: pnpm --filter @pb/api db:generate

      - name: Typecheck (shared/api/web)
        run: pnpm typecheck

      - name: Typecheck mobile
        run: pnpm typecheck:mobile

      - name: API tests
        run: pnpm --filter @pb/api test

      - name: Build web
        run: pnpm --filter @pb/web build

      - name: Upload web build
        uses: actions/upload-artifact@v4
        with:
          name: web-dist
          path: v2/apps/web/dist
          retention-days: 7

  docker:
    name: Docker build (api + web)
    runs-on: ubuntu-latest
    needs: verify
    steps:
      - uses: actions/checkout@v4
      - name: Build API image
        run: docker build -f v2/apps/api/Dockerfile -t pb-api:ci v2
      - name: Build web image
        run: docker build -f v2/apps/web/Dockerfile -t pb-web:ci v2
